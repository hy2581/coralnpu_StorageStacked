cmake_minimum_required(VERSION 3.20)
project(coralnpu_StorageStacked LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-UNDEBUG)
set(STORAGE_STACK_ROOT "$ENV{STORAGE_STACK_ROOT}" CACHE PATH "Independent AXI storage source")
if(NOT STORAGE_STACK_ROOT)
  set(STORAGE_STACK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../axi_StorageStacked")
endif()
set(STORAGE_BUILD_CLI OFF CACHE BOOL "" FORCE)
set(STORAGE_MEMSIM_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/build/memsim/libstoragestacked_memsim.so" CACHE FILEPATH "" FORCE)
add_subdirectory(${STORAGE_STACK_ROOT} storage)
add_executable(coralnpu_sim simulation/main.cc simulation/axi_master.cc simulation/axi_demo.cc)
target_include_directories(coralnpu_sim PRIVATE integration/coralnpu)
target_link_libraries(coralnpu_sim PRIVATE StorageStacked::axi
  ${CMAKE_CURRENT_SOURCE_DIR}/build/coralnpu/libcoralnpu-native.so)
